# Dependências extras pra rodar LOCALMENTE no Windows com MetaTrader 5.
#
# NÃO suba este arquivo como requirements.txt do Streamlit Cloud — o
# pacote MetaTrader5 é uma DLL do Windows e a instalação vai FALHAR no
# servidor Linux da nuvem, quebrando o deploy. Use assim, na sua
# máquina Windows:
#
#   pip install -r requirements.txt
#   pip install -r requirements-local.txt
#
MetaTrader5>=5.0.45

# Trava a versão do NumPy abaixo de 2.1 nesta máquina. A partir da
# 2.1, os wheels pré-compilados do NumPy pra Windows passaram a exigir
# instruções de CPU "x86-64-v2" (inclui SSE4.2) — em processadores
# mais antigos (ou algumas VMs) sem esse suporte, o import quebra com
# "RuntimeError: NumPy was built with baseline optimizations (X86_V2)
# but your machine doesn't support (X86_V2)". A 1.26.4 é a última
# versão 1.x com suporte oficial ao Python 3.12+ e não tem essa
# exigência — funciona com o baseline SSE2, presente em qualquer CPU
# x86-64. Sem esse pin, o próximo "pip install -r requirements.txt"
# (inclusive dentro do workflow do GitHub Actions) reinstalaria a
# versão mais nova e quebraria de novo.
numpy==1.26.4
